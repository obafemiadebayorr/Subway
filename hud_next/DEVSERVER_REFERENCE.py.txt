_token = {"value": "", "at": 0.0}


def hud_token(force=False):
    if force or not _token["value"] or time.time() - _token["at"] > 300:
        html = urllib.request.urlopen(HUD + "/", timeout=5).read().decode("utf-8", "replace")
        m = re.search(r'name="subway-token" content="([^"]+)"', html)
        _token["value"], _token["at"] = (m.group(1) if m else ""), time.time()
    return _token["value"]


# Workers page: safe worker fields only (never prompt, secret, env or paths).
WK_KEYS = ("id", "role", "status", "provider", "model", "effort", "track", "created_at", "started_at", "completed_at",
           "exit_code", "isolate", "branch_name", "merged", "launched_by", "requested_by", "paid_by", "mode",

def _wk(w):
    o = {k: w.get(k) for k in WK_KEYS if w.get(k) not in (None, "")}
    u = w.get("usage") or {}
    if isinstance(u, dict):
        o["tokens"] = int(u.get("input") or 0) + int(u.get("output") or 0) or int(u.get("total") or 0)
    if isinstance(w.get("review"), (str, bool)):
        o["review"] = w["review"]
    for src in (w.get("budget"), w.get("spin_args")):
        if isinstance(src, dict) and str(src.get("max_tokens") or "").isdigit() and int(src["max_tokens"]) > 0:
            o["cap"] = int(src["max_tokens"])
            break
    sa = w.get("spin_args") if isinstance(w.get("spin_args"), dict) else {}
    bm = (w.get("billing_mode") or sa.get("billing_mode") or "")
    if str(bm).upper() in ("A", "B", "C"):
        o["billing_mode"] = str(bm).upper()
    r = w.get("report")
    if isinstance(r, str) and r.strip():
        o["report"] = r[-1500:]
    bs = w.get("borrowed_skills") or sa.get("skills") or []
    if isinstance(bs, list) and bs:
        o["skills"] = [str(x.get("name") if isinstance(x, dict) else x)[:80] for x in bs][:8]
    nd = sa.get("needs") or w.get("needs") or []
    if isinstance(nd, list) and nd:
        o["mcp"] = [str(x)[:60] for x in nd][:8]
    return o


def hud_call(path, body=None):
    for attempt in (0, 1):
        req = urllib.request.Request(HUD + path, method="POST" if body is not None else "GET",
                                     data=json.dumps(body).encode() if body is not None else None)
        req.add_header("X-Subway-Token", hud_token(force=attempt == 1))
        if body is not None:
            req.add_header("Content-Type", "application/json")
            req.add_header("Origin", HUD)  # same-origin to the HUD, which its settings guard requires
        try:
            with urllib.request.urlopen(req, timeout=10) as r:
                return r.status, r.read()
        except urllib.error.HTTPError as e:
            if e.code == 401 and attempt == 0:
                continue  # token rotated: fetch it again once
            return e.code, e.read()
    return 502, b'{"error":"HUD refused the token"}'


class H(http.server.SimpleHTTPRequestHandler):
    def __init__(self, *a, **k):
        super().__init__(*a, directory=ROOT, **k)

    def log_message(self, *a):
        pass

    def end_headers(self):
        self.send_header("Cache-Control", "no-store")
        super().end_headers()

    def _host_ok(self):
        if (self.headers.get("Host") or "") in HOSTS:
            return True
        self.send_error(403)
        return False

    def _json(self, status, raw):
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(raw)))
        self.end_headers()
        self.wfile.write(raw)

    def do_HEAD(self):
        if self._host_ok():
            super().do_HEAD()

    def _stream(self):
        """SSE passthrough (never buffered): forwards each line from the HUD the moment it arrives."""
        req = urllib.request.Request(HUD + self.path)
        req.add_header("X-Subway-Token", hud_token())
        if self.headers.get("Last-Event-ID"):
            req.add_header("Last-Event-ID", self.headers["Last-Event-ID"])
        try:
            up = urllib.request.urlopen(req, timeout=45)  # the HUD sends a keep-alive at least every 15s
        except urllib.error.HTTPError as e:
            if e.code == 401:
                hud_token(force=True)
            return self._json(e.code, e.read())
        except Exception as e:
            return self._json(502, json.dumps({"error": f"HUD unreachable: {e.__class__.__name__}"}).encode())
        self.send_response(200)
        self.send_header("Content-Type", "text/event-stream")
        self.send_header("X-Accel-Buffering", "no")
        self.end_headers()
        try:
            with up:
                for line in iter(up.readline, b""):
                    self.wfile.write(line)
                    if line in (b"\n", b"\r\n"):
                        self.wfile.flush()
        except Exception:
            pass  # browser left or HUD restarted; EventSource reconnects
        self.close_connection = True

    def do_GET(self):
        if not self._host_ok():
            return
        if self.path.split("?", 1)[0] in ("/api/events", "/api/chat_stream"):
            return self._stream()
        path = self.path.split("?", 1)[0]
        if re.fullmatch(r"/hud/[a-z0-9_-]+\.(js|css)", path):
            # The HUD's own modules (read-only), so Usage and Stats render with the real code.
            f = os.path.join(ROOT, path[5:]) if path[5:] in ("stats.js", "stats.css", "terrain.js") else ""  # proto overrides
            if not os.path.isfile(f):
                f = os.path.join(HUD_DIR, path[5:])
            if not os.path.isfile(f):
                return self.send_error(404)
            raw = open(f, "rb").read()
            self.send_response(200)
            self.send_header("Content-Type", "text/css; charset=utf-8" if f.endswith(".css") else "text/javascript; charset=utf-8")
            self.send_header("Content-Length", str(len(raw)))
            self.end_headers()
            return self.wfile.write(raw)
        if path.startswith("/api/"):
            if path not in ("/api/remote_workers", "/api/settings", "/api/status", "/api/wake", "/api/wake_rules", "/api/fastmail",
                            "/api/stats", "/api/worker_activity", "/api/diagnostics", "/api/worker_digest",
                            "/api/pair/list", "/api/station", "/api/ledger", "/api/token_savings", "/api/metrics",
                            "/api/collab/session", "/api/collab/usage", "/api/collab/people",
                            "/api/peers", "/api/peers/train", "/api/peers/allowance",
                            "/api/peers/allowance/history", "/api/peers/models", "/api/coordination",
                            "/api/peer/coordination",
                            "/api/terminals", "/api/terminal", "/api/pending_runs",
                            "/api/landing", "/api/chats", "/api/chat_read", "/api/ping",
                            "/api/mcp/connect", "/api/mcp/hub/servers", "/api/skills", "/api/usage/ledger", "/api/coordination/all", "/api/screen/active", "/api/screen/session", "/api/worker_log", "/api/audit"):
                return self._json(404, b'{"error":"Not proxied by the prototype"}')
            try:
                status, raw = hud_call(self.path)
            except Exception as e:
                return self._json(502, json.dumps({"error": f"HUD unreachable at {HUD}: {e.__class__.__name__}"}).encode())
            if path == "/api/status" and status == 200:
                # Only what the prototype shows, so the browser never holds the whole status payload.
                data = json.loads(raw or b"{}")
                st = data.get("settings") or {}
                raw = json.dumps({"capacity": st.get("capacity") or {},
                                  "settings": {k: st[k] for k in READ if k in st},
                                  "workers": [_wk(w) for w in (data.get("workers") or [])],
                                  "history": [_wk(w) for w in (data.get("history") or [])],
                                  "total_history": data.get("total_history") or 0,
                                  # Usage page: limits per tool and the Copilot inline estimate (counts only, no keys).
                                  "quotas": data.get("quotas") or {},
                                  "copilot_inline_estimate": data.get("copilot_inline_estimate") or {},
                                  "timestamp": data.get("timestamp"),
                                  # Models tab: ranking track and models out of allowance.
                                  "models_track": data.get("models_track"),
                                  "exhausted_models": data.get("exhausted_models"),
                                  # Workers page: leases, sign-offs and decision tracks.
                                  "locks": data.get("locks") or {}, "signoffs": data.get("signoffs") or {},
                                  "tracks": data.get("tracks") or {}}).encode()
            return self._json(status, raw)
        super().do_GET()

    def do_POST(self):
        if not self._host_ok():
            return
        if self.path != "/api/settings" and self.path not in EXTRA_POST:
            return self._json(404, b'{"error":"Not proxied by the prototype"}')
        if (self.headers.get("Origin") or "") not in ORIGINS:
            return self._json(403, b'{"error":"Origin not allowed"}')
        try:
            body = json.loads(self.rfile.read(int(self.headers.get("Content-Length") or 0)) or b"{}")
        except ValueError:
            return self._json(400, b'{"error":"Bad JSON"}')
        if self.path in EXTRA_POST:
            if not isinstance(body, dict):
                return self._json(400, b'{"error":"Expected a JSON object"}')
            try:
                status, raw = hud_call(self.path, body)
            except Exception as e:
                return self._json(502, json.dumps({"error": f"HUD unreachable at {HUD}: {e.__class__.__name__}"}).encode())
            if self.path == "/api/pair/approve":
                # The new device's bearer token is for the phone only; the browser never holds it.
                try:
                    out = json.loads(raw or b"{}")
                    out.pop("token", None) if isinstance(out, dict) else None
                    raw = json.dumps(out).encode()
                except ValueError:
                    raw = b'{"error":"Bad reply from HUD"}'
            return self._json(status, raw)
        if not isinstance(body, dict) or not body or set(body) - WRITABLE:
            return self._json(400, json.dumps({"error": f"Only worker settings can be changed here: {sorted(set(body) - WRITABLE) if isinstance(body, dict) else body}"}).encode())
        try:
            status, raw = hud_call("/api/settings", body)
        except Exception as e:
            return self._json(502, json.dumps({"error": f"HUD unreachable at {HUD}: {e.__class__.__name__}"}).encode())
        return self._json(status, raw)


if __name__ == "__main__":
    http.server.ThreadingHTTPServer(("127.0.0.1", PORT), H).serve_forever()
